from functools import lru_cache

def tsp_with_path(distance):
    n = len(distance)

    @lru_cache(maxsize=None)
    def dp(mask, pos):
        if mask == (1 << n) - 1:
            return distance[pos][0], [0]

        best_cost = float('inf')
        best_path = []

        for city in range(n):
            if not (mask & (1 << city)):
                cost, path = dp(mask | (1 << city), city)
                cost += distance[pos][city]

                if cost < best_cost:
                    best_cost = cost
                    best_path = [city] + path

        return best_cost, best_path

    cost, path = dp(1, 0)
    return cost, [0] + path


graph = [
    [0, 10, 15, 20],
    [10, 0, 35, 25],
    [15, 35, 0, 30],
    [20, 25, 30, 0]
]

cost, path = tsp_with_path(graph)

print("Minimum Cost:", cost)
print("Tour:", " -> ".join(map(str, path)))


#Complexity

#Time: O(n² · 2ⁿ)
#Space: O(n · 2ⁿ)
